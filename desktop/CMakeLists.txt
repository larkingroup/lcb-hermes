cmake_minimum_required(VERSION 3.20)
project(lcb_hermes VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_library(hermes_core STATIC src/core.c src/markdown.c vendor/cjson/cJSON.c)
target_include_directories(hermes_core PUBLIC src vendor/cjson)
target_compile_definitions(hermes_core PRIVATE CJSON_NESTING_LIMIT=32)
add_executable(core_test tests/core_test.c)
target_link_libraries(core_test PRIVATE hermes_core)
add_test(NAME protocol_and_chats COMMAND core_test)
add_executable(markdown_test tests/markdown_test.c)
target_link_libraries(markdown_test PRIVATE hermes_core)
add_test(NAME markdown COMMAND markdown_test)
if(WIN32)
    enable_language(RC)
    add_library(hermes_windows STATIC src/transport_win.c src/store_win.c src/classic_win.c)
    target_link_libraries(hermes_windows PUBLIC hermes_core winhttp ws2_32 crypt32 shell32 ole32 uuid comctl32 gdi32 user32)
    target_compile_definitions(hermes_windows PUBLIC _WIN32_WINNT=0x0a00)
    add_executable(lcb-hermes WIN32 src/winmain.c assets/hermes.rc)
    target_include_directories(lcb-hermes PRIVATE assets)
    target_link_libraries(lcb-hermes PRIVATE hermes_windows comdlg32 dwmapi windowscodecs shlwapi)
    add_executable(windows_test tests/windows_test.c)
    target_link_libraries(windows_test PRIVATE hermes_windows)
    add_test(NAME endpoint_and_private_storage COMMAND windows_test)
    add_executable(desktop_test tests/desktop_test.c assets/hermes.rc)
    target_include_directories(desktop_test PRIVATE assets)
    target_link_libraries(desktop_test PRIVATE hermes_windows comdlg32 dwmapi windowscodecs shlwapi)
    add_test(NAME desktop_regressions COMMAND desktop_test)
    set_tests_properties(desktop_regressions PROPERTIES TIMEOUT 30)
    add_executable(live_test tests/live_test.c)
    target_link_libraries(live_test PRIVATE hermes_windows)
    add_executable(desktop_live_test tests/desktop_live_test.c assets/hermes.rc)
    target_include_directories(desktop_live_test PRIVATE assets)
    target_link_libraries(desktop_live_test PRIVATE hermes_windows comdlg32 dwmapi windowscodecs shlwapi)
endif()
get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/sdl;/GS;/guard:cf;/utf-8>")
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        if(target MATCHES "_test$")
            target_compile_options(${target} PRIVATE /UNDEBUG)
        endif()
        target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror;-fstack-protector-strong>")
        if(target MATCHES "_test$")
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
        if(target STREQUAL "lcb-hermes" AND WIN32)
            target_link_options(${target} PRIVATE -municode -Wl,--dynamicbase,--nxcompat)
        endif()
    endif()
endforeach()
