cmake_minimum_required(VERSION 3.20)
project(lcb_hermes VERSION 0.50.0 LANGUAGES C)
set(LCB_RELEASE_VERSION "0.50-pre1")
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(hermes_core STATIC src/core.c src/markdown.c vendor/cjson/cJSON.c)
target_include_directories(hermes_core PUBLIC src vendor/cjson)
target_compile_definitions(hermes_core PRIVATE CJSON_NESTING_LIMIT=32)
if(WIN32)
    enable_language(RC)
    add_library(hermes_windows STATIC src/transport_win.c src/store_win.c src/classic_win.c)
    target_link_libraries(hermes_windows PUBLIC hermes_core winhttp ws2_32 crypt32 shell32 ole32 uuid comctl32 gdi32 user32 uxtheme)
    target_compile_definitions(hermes_windows PUBLIC _WIN32_WINNT=0x0a00)
    add_executable(lcb-hermes WIN32 src/winmain.c assets/hermes.rc)
    target_include_directories(lcb-hermes PRIVATE assets)
    target_link_libraries(lcb-hermes PRIVATE hermes_windows comdlg32 dwmapi windowscodecs shlwapi)
endif()
set(LCB_LINUX_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(LCB_LINUX_DEFAULT ON)
endif()
option(LCB_BUILD_LINUX "Build the small Xt/Athena Linux client" ${LCB_LINUX_DEFAULT})
if(LCB_BUILD_LINUX AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LINUX_DEPS REQUIRED IMPORTED_TARGET xaw7 xt x11 libsoup-3.0 libsecret-1 libcrypto)
    add_library(hermes_linux STATIC ${CMAKE_CURRENT_SOURCE_DIR}/../linux/src/transport.c ${CMAKE_CURRENT_SOURCE_DIR}/../linux/src/store.c)
    target_include_directories(hermes_linux PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../linux/src)
    target_link_libraries(hermes_linux PUBLIC hermes_core PkgConfig::LINUX_DEPS)
    add_executable(lcb-hermes ${CMAKE_CURRENT_SOURCE_DIR}/../linux/src/main.c)
    target_link_libraries(lcb-hermes PRIVATE hermes_linux)
    target_link_options(lcb-hermes PRIVATE -Wl,--as-needed)
    if(CMAKE_BUILD_TYPE STREQUAL "MinSizeRel")
        target_compile_options(hermes_core PRIVATE -ffunction-sections -fdata-sections)
        target_compile_options(hermes_linux PRIVATE -ffunction-sections -fdata-sections)
        target_compile_options(lcb-hermes PRIVATE -ffunction-sections -fdata-sections)
        target_link_options(lcb-hermes PRIVATE -Wl,--gc-sections -Wl,-s)
    endif()
    install(TARGETS lcb-hermes RUNTIME DESTINATION bin)
endif()
get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/sdl;/GS;/guard:cf;/utf-8>")
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror;-fstack-protector-strong>")
        if(target STREQUAL "lcb-hermes" AND WIN32)
            target_link_options(${target} PRIVATE -municode -Wl,--dynamicbase,--nxcompat)
        endif()
    endif()
endforeach()
option(LCB_BUILD_TESTS "Build native contract regression checks" OFF)
if(LCB_BUILD_TESTS AND WIN32)
    enable_testing()
    add_executable(contracts_test tests/contracts_test.c assets/hermes.rc)
    target_link_libraries(contracts_test PRIVATE hermes_windows comdlg32 dwmapi windowscodecs shlwapi)
    if(MSVC)
        target_compile_options(contracts_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
    else()
        target_compile_options(contracts_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME hermes_contracts COMMAND contracts_test)
    set_tests_properties(hermes_contracts PROPERTIES TIMEOUT 30)
    add_executable(server_panel_test tests/server_panel_test.c assets/hermes.rc)
    target_link_libraries(server_panel_test PRIVATE hermes_windows comdlg32 dwmapi windowscodecs shlwapi)
    if(MSVC)
        target_compile_options(server_panel_test PRIVATE /W4 /WX /UNDEBUG /utf-8)
    else()
        target_compile_options(server_panel_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_test(NAME server_management_contracts COMMAND server_panel_test)
    set_tests_properties(server_management_contracts PROPERTIES TIMEOUT 30)
endif()
if(LCB_BUILD_TESTS AND TARGET hermes_linux)
    enable_testing()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(check core store transport ui)
        add_executable(linux_${check}_test ${CMAKE_CURRENT_SOURCE_DIR}/../linux/tests/${check}_test.c)
        target_link_libraries(linux_${check}_test PRIVATE hermes_linux)
        target_compile_options(linux_${check}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endforeach()
    add_test(NAME linux_core COMMAND linux_core_test)
    add_test(NAME linux_encrypted_store COMMAND linux_store_test)
    add_test(NAME linux_transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../linux/tests/run_transport.py $<TARGET_FILE:linux_transport_test>)
    find_program(XVFB_EXECUTABLE Xvfb REQUIRED)
    add_test(NAME linux_ui COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../linux/tests/run_ui.py $<TARGET_FILE:linux_ui_test> ${XVFB_EXECUTABLE})
    set_tests_properties(linux_core linux_encrypted_store linux_transport linux_ui PROPERTIES TIMEOUT 45)
endif()
